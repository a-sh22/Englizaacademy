/* =========================================
   NORMALIZE TEXT
========================================= */

function normalizeText(text) {
    return text
        .toLowerCase()
        .trim()
        .replace(/’/g,"'")
        .replace(/\bi'm\b/g,"i am")
        .replace(/\bi've\b/g,"i have")
        .replace(/\bi'd\b/g,"i would")
        .replace(/\bi'll\b/g,"i will")
        .replace(/\bhe's\b/g,"he is")
        .replace(/\bshe's\b/g,"she is")
        .replace(/\bit's\b/g,"it is")
        .replace(/\byou're\b/g,"you are")
        .replace(/\bwe're\b/g,"we are")
        .replace(/\bthey're\b/g,"they are")
        .replace(/\bhasn't\b/g,"has not")
        .replace(/\bhaven't\b/g,"have not")
        .replace(/\bisn't\b/g,"is not")
        .replace(/\baren't\b/g,"are not")
        .replace(/\bwasn't\b/g,"was not")
        .replace(/\bweren't\b/g,"were not")
        .replace(/\bdon't\b/g,"do not")
        .replace(/\bdoesn't\b/g,"does not")
        .replace(/\bdidn't\b/g,"did not")
        .replace(/\bcan't\b/g,"cannot")
        .replace(/\bwon't\b/g,"will not")
        .replace(/[.,!?؟]/g,"")
        .replace(/'/g,"")
        .replace(/-/g," ")
        .replace(/\s+/g," ")
        .trim();
}

/* =========================================
   WRITING ANSWERS
========================================= */

function checkWriting(inputId,resultId,correctAnswers) {
    let input = document.getElementById(inputId);
    let result = document.getElementById(resultId);

    let userAnswer = normalizeText(input.value);
    let answers = correctAnswers.map(answer => normalizeText(answer));

    input.classList.remove("correct-input","wrong-input","shake");

    if (answers.includes(userAnswer)) {
        input.classList.add("correct-input");

        result.innerHTML =
            '<div class="confetti">'+
            '<span class="star">✨</span>'+
            '<span class="star">⭐</span>'+
            '<span class="star">🌟</span>'+
            '<span class="star">✨</span>'+
            '<span class="star">⭐</span>'+
            '</div>'+
            '✅ أحسنت! لقد كتبت الإجابة الصحيحة.';

        result.className = "answer-result success-text";
    } else {
        input.classList.add("wrong-input","shake");

        setTimeout(function() {
            input.classList.remove("shake");
        },300);

        result.innerHTML =
            "😅❌ إجابتك غير صحيحة.<br><br>✅ إحدى الإجابات الصحيحة هي:<br><br><b>"+
            correctAnswers[0]+
            "</b>";

        result.className = "answer-result error-text";
    }
}

/* =========================================
   AGE ANSWER
========================================= */

function checkAgeAnswer(inputId,resultId) {
    let input = document.getElementById(inputId);
    let result = document.getElementById(resultId);

    let userAnswer = normalizeText(input.value);

    input.classList.remove("correct-input","wrong-input","shake");

    let agePattern = /^i am \d+( years old)?$/;

    if (agePattern.test(userAnswer)) {
        input.classList.add("correct-input");

        result.innerHTML =
            '<div class="confetti">'+
            '<span class="star">✨</span>'+
            '<span class="star">⭐</span>'+
            '<span class="star">🌟</span>'+
            '<span class="star">✨</span>'+
            '<span class="star">⭐</span>'+
            '</div>'+
            '✅ أحسنت! لقد أجبت عن السؤال بطريقة صحيحة.';

        result.className = "answer-result success-text";
    } else {
        input.classList.add("wrong-input","shake");

        setTimeout(function() {
            input.classList.remove("shake");
        },300);

function checkAnswer(button,isCorrect,resultId) {
    let result = document.getElementById(resultId);

    let allButtons = button.parentElement.querySelectorAll(".choice-btn");

    allButtons.forEach(btn => {
        btn.disabled = true;
    });

    if (isCorrect) {
        button.classList.add("correct");
        button.innerHTML += " ✔️";

        result.innerHTML =
            '<div class="confetti">'+
            '<span class="star">✨</span>'+
            '<span class="star">⭐</span>'+
            '<span class="star">🌟</span>'+
            '<span class="star">✨</span>'+
            '<span class="star">⭐</span>'+
            '</div>'+
            '✅ أحسنت! لقد اخترت الإجابة الصحيحة.';

        result.className = "result-text success-text";
    } else {
        button.classList.add("wrong","shake");

        setTimeout(function() {
            button.classList.remove("shake");
        },300);

        button.innerHTML += " ✖️";

        result.innerHTML = "😅❌ لقد اخترت الإجابة الخاطئة.";
        result.className = "result-text error-text";

        allButtons.forEach(btn => {
            if (btn.getAttribute("onclick").includes("true")) {
                btn.classList.add("correct");

                if (!btn.innerHTML.includes("✔️")) {
                    btn.innerHTML += " ✔️";
                }
            }
        });
    }
}

function addWord(button,answerBoxId) {
    if (button.style.display === "none") return;

    let answerBox = document.getElementById(answerBoxId);
    let newWord = document.createElement("div");

    newWord.className = "selected-word";
    newWord.innerText = button.innerText;

    answerBox.appendChild(newWord);
    button.style.display = "none";
}

function resetSentence(wordsId,answerBoxId) {
    let container = document.getElementById(wordsId);

    container.querySelectorAll(".word-btn").forEach(button => {
        button.style.display = "inline-block";
    });

    document.getElementById(answerBoxId).innerHTML = "";
}

function checkSentence(answerBoxId,correctSentence,resultId) {
    let answerBox = document.getElementById(answerBoxId);
    let result = document.getElementById(resultId);

    let words = answerBox.querySelectorAll(".selected-word");
    let userSentence = "";

    words.forEach(word => {
        userSentence += word.innerText + " ";
    });

    userSentence = normalizeText(userSentence);
    correctSentence = normalizeText(correctSentence);

    answerBox.classList.remove("shake");

    if (userSentence === correctSentence) {
        result.innerHTML =
            '<div class="confetti">'+
            '<span class="star">✨</span>'+
            '<span class="star">⭐</span>'+
            '<span class="star">🌟</span>'+
            '<span class="star">✨</span>'+
            '<span class="star">⭐</span>'+
            '</div>'+
            '✅ أحسنت! لقد رتبت الكلمات بطريقة صحيحة.';

        result.className = "result-text success-text";
        answerBox.style.borderColor = "#8fd19e";
        answerBox.style.background = "#eefdf3";
    } else {
        answerBox.classList.add("shake");

        setTimeout(function() {
            answerBox.classList.remove("shake");
        },300);

        answerBox.style.borderColor = "#ffb8b8";
        answerBox.style.background = "#fff2f2";

        result.innerHTML = "😅❌ ترتيب الكلمات غير صحيح.";
        result.className = "result-text error-text";
    }
}

function shuffleWords(containerId) {
    let container = document.getElementById(containerId);

    if (!container) return;

    let buttons = Array.from(container.querySelectorAll(".word-btn"));

    buttons.sort(() => Math.random() - 0.5);

    container.innerHTML = "";

    buttons.forEach(button => {
        container.appendChild(button);
    });
}

function toggleExamples() {
    let box = document.getElementById("examplesBox");
    if (!box) return;

    let btn = box.previousElementSibling;

    if (box.style.display === "none") {
        box.style.display = "block";
        btn.innerHTML = "📕 إخفاء الجمل";
    } else {
        box.style.display = "none";
        btn.innerHTML = "📖 عرض الجمل";
    }
}

/* =========================================
   ESCAPE HTML
========================================= */

function escapeHTML(value) {
    return String(value)
        .replace(/&/g,"&amp;")
        .replace(/</g,"&lt;")
        .replace(/>/g,"&gt;")
        .replace(/"/g,"&quot;")
        .replace(/'/g,"&#039;");
}

/* =========================================
   CREATE VOCABULARY ITEM
========================================= */

function createVocabularyItem(item) {
    return `
        <article class="vocab-item">

            <div class="vocab-word-row">
                <div class="vocab-word">${escapeHTML(item.word)}</div>

                <button
                    class="pronounce-btn"
                    type="button"
                    data-word="${escapeHTML(item.word)}"
                    aria-label="Listen to pronunciation"
                    title="Listen to pronunciation"
                >🔊</button>
            </div>

            ${item.pronunciation ? `
                <div class="pronunciation-info">
                    <span class="pronunciation-ipa">
                        ${escapeHTML(item.pronunciation)}
                    </span>

                    ${item.pronunciationArabic ? `
                        <span class="pronunciation-ar">
                            ${escapeHTML(item.pronunciationArabic)}
                        </span>
                    ` : ""}
                </div>
            ` : ""}

            <div class="vocab-meaning">
                ${escapeHTML(item.meaning)}
            </div>

            <span class="word-type">
                ${escapeHTML(item.type)}
            </span>

            <div class="examples">
                <div class="example">
                    <div class="example-en">
                        ${escapeHTML(item.example)}
                    </div>

                    <div class="example-ar">
                        ${escapeHTML(item.translation)}
                    </div>
                </div>
            </div>

            ${item.grammarNote ? `
                <div class="grammar-note">
                    <strong>ملاحظة نحوية</strong>
                    <p>${escapeHTML(item.grammarNote)}</p>
                </div>
            ` : ""}

        </article>
    `;
}

/* =========================================
   RENDER VOCABULARY LISTS
========================================= */

function renderVocabulary() {
    Object.entries(readingVocabulary).forEach(([containerId,keys]) => {
        const container = document.getElementById(containerId);

        if (!container) return;

        container.innerHTML = keys
            .map(key => vocabulary[key])
            .filter(Boolean)
            .map(createVocabularyItem)
            .join("");
    });
}

function toggleVocabulary(button) {
    const section = button.closest(".vocab-section");
    const isOpen = section.classList.toggle("open");

    button.setAttribute("aria-expanded",String(isOpen));

    const label = button.querySelector(".vocab-action-label");

    if (label) {
        label.textContent = isOpen
            ? "إخفاء المفردات"
            : "اضغط لعرض المفردات";
    }
}

/* =========================================
   WORD PRONUNCIATION
========================================= */

let englishVoice = null;

function loadEnglishVoice() {
    if (!("speechSynthesis" in window)) return;

    const voices = window.speechSynthesis.getVoices();

    englishVoice =
        voices.find(voice => voice.lang === "en-US") ||
        voices.find(voice => voice.lang.startsWith("en-")) ||
        null;
}

if ("speechSynthesis" in window) {
    loadEnglishVoice();
    window.speechSynthesis.onvoiceschanged = loadEnglishVoice;
}

function speakWord(word) {
    if (!("speechSynthesis" in window)) {
        alert("عذرًا، متصفحك لا يدعم تشغيل النطق.");
        return;
    }

    if (!word) return;

    const speech = window.speechSynthesis;

    speech.cancel();
    speech.resume();

    const utterance = new SpeechSynthesisUtterance(word);

    utterance.lang = "en-US";
    utterance.rate = 0.85;
    utterance.pitch = 1;

    if (!englishVoice) {
        loadEnglishVoice();
    }

    if (englishVoice) {
        utterance.voice = englishVoice;
    }

    speech.speak(utterance);
}

document.addEventListener("click",function(event) {
    const button = event.target.closest(".pronounce-btn");

    if (!button) return;

    speakWord(button.getAttribute("data-word"));
});
